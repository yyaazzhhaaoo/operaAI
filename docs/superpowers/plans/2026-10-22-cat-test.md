```python
# -*- coding: utf-8 -*-
"""摸底题库种子：从真实分段里选 6 题写入 cat_questions。

用法（项目根目录）：
    .venv/bin/python scripts/seed_cat_questions.py

幂等：按 seq 先删后插。题目定位不写死 id，按标题找带段音频的分段；
有多条时取 demo_id 最大的那条（最近一次上传）。
"""

import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

from sqlalchemy import delete, select                      # noqa: E402

from app.db import session_scope                           # noqa: E402
from app.models import CatQuestion, Segment                # noqa: E402

# (seq, 分段标题, focus_dims)
QUESTIONS = [
    (1, "辕门外三声炮如同雷震", ["音准", "气息", "拖腔"]),
    (2, "天波府里走出来我保国臣", ["音准", "节奏", "咬字"]),
    (3, "头戴金冠压双鬓", ["节奏", "咬字"]),
    (4, "帅字旗飘入云", ["气息", "拖腔"]),
    (5, "斗大的穆字震乾坤", ["音准", "咬字"]),
    (6, "谁料想我五十三岁又管三军", ["节奏", "气息", "拖腔"]),
]


def _find_segment(db, title: str) -> Segment | None:
    return db.scalar(
        select(Segment)
        .where(Segment.title == title)
        .where(Segment.audio_id.is_not(None))
        .where(Segment.demo_id.is_not(None))
        .order_by(Segment.demo_id.desc(), Segment.id.desc())
        .limit(1)
    )


def main() -> int:
    with session_scope() as db:
        rows = []
        missing = []
        for seq, title, dims in QUESTIONS:
            seg = _find_segment(db, title)
            if seg is None:
                missing.append(title)
                continue
            rows.append((seq, seg, dims))

        if missing:
            print("找不到带段音频的分段：" + "、".join(missing))
            return 1

        db.execute(delete(CatQuestion).where(
            CatQuestion.seq.in_([q[0] for q in QUESTIONS])
        ))
        for seq, seg, dims in rows:
            db.add(CatQuestion(
                seq=seq,
                segment_id=seg.id,
                focus_dims=dims,
                is_active=True,
            ))
            print(f"seq={seq} segment_id={seg.id} {seg.title}")

    print(f"已写入 {len(rows)} 题")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
# app/repositories/cat_questions_repo.py
from sqlalchemy import select
from sqlalchemy.orm import Session, selectinload

from app.models import CatQuestion, Segment


def list_active(db: Session) -> list[CatQuestion]:
    """启用中的题目，按 seq 排序；一次取回 segment 与 demo，避免 N+1。"""
    return list(db.scalars(
        select(CatQuestion)
        .options(selectinload(CatQuestion.segment).selectinload(Segment.demo))
        .where(CatQuestion.is_active.is_(True))
        .where(CatQuestion.segment_id.is_not(None))
        .order_by(CatQuestion.seq)
    ).all())


def get_active_by_segment(db: Session, segment_id: int) -> CatQuestion | None:
    """某分段是否是启用中的摸底题；J2 用它挡住普通练习冒充摸底。"""
    return db.scalar(
        select(CatQuestion)
        .where(CatQuestion.segment_id == segment_id)
        .where(CatQuestion.is_active.is_(True))
        .limit(1)
    )
```

```python
            student_id=metadata["student_id"],
            segment_id=metadata["segment_id"],

            dimensions_json=result.get("dimensions"),
            words_json=result.get("words"),
            cdm_json=result.get("cdm"),
            duration_sec=metadata.get("duration_sec"),
```

```python
# -*- coding: utf-8 -*-
"""J1–J3 摸底测试与 8.10 跳过的业务逻辑。

J2 复用 B2 分析管线，落 practice_records(source='cat_test') 并更新 BKT；
J3 的 record_id 只作学生锚点，报告返回该生最新画像（口径见 DOC_ISSUES 第 47 条）。
"""

from sqlalchemy.orm import Session

from app.common.errors import BusinessError
from app.models import BktHistory, PracticeRecord
from app.repositories import (
    audio_repo,
    bkt_history_repo,
    cat_questions_repo,
    user_repo,
)
from app.schemas.audio import AnalyzeSubmitIn
from app.schemas.cat import CatSkillOut
from app.services import analyze_service

PROFILE_SKILLS = ("音准控制", "气息支撑", "拖腔", "归韵咬字", "节奏感知")
_SKIP_P_L = 0.35
_SKIP_CONFIDENCE = 0.4


def _own_student_id(db: Session, user_id: int) -> int:
    student_id = user_repo.get_student_id_by_user_id(db, user_id)
    if student_id is None:
        raise BusinessError(403, "当前账号没有学生档案")
    return student_id


def _lyrics_text(lyrics_json) -> str:
    if not isinstance(lyrics_json, list):
        return ""
    return "".join(
        str(item.get("word", ""))
        for item in lyrics_json
        if isinstance(item, dict)
    )


def questions(db: Session) -> list[dict]:
    """J1：启用中的题目，按 seq 排序。audio_url 由 API 层补。"""
    out = []
    for q in cat_questions_repo.list_active(db):
        segment = q.segment
        if segment is None:
            continue
        out.append({
            "id": q.id,
            "seq": q.seq,
            "segment_id": segment.id,
            "title": segment.title or f"第 {q.seq} 句",
            "demo_title": segment.demo.title if segment.demo else "",
            "duration": segment.duration,
            "focus_dims": q.focus_dims if isinstance(q.focus_dims, list) else [],
            "lyrics": _lyrics_text(segment.lyrics_json),
            "audio_id": segment.audio_id,
        })
    return out


def submit(
    db: Session,
    *,
    user_id: int,
    segment_id: int,
    audio_id: int,
    duration_sec: float | None,
) -> str:
    """J2：校验归属与题目后把任务投进 B2 管线。"""
    student_id = _own_student_id(db, user_id)

    audio = audio_repo.get_by_id(db, audio_id)
    if audio is None:
        raise BusinessError(404, "音频不存在")
    if audio.uploader_id != user_id:
        raise BusinessError(403, "只能提交本人录音")

    if cat_questions_repo.get_active_by_segment(db, segment_id) is None:
        raise BusinessError(404, "题目不存在")

    return analyze_service.submit(
        db,
        AnalyzeSubmitIn(segment_id=segment_id, student_audio_id=audio_id),
        persist={
            "kind": "cat_test",
            "student_id": student_id,
            "segment_id": segment_id,
            "audio_id": audio_id,
            "duration_sec": duration_sec,
        },
    )


def _profile(db: Session, student_id: int) -> list[CatSkillOut]:
    latest = {
        row.skill: row
        for row in bkt_history_repo.get_latest_per_skill_for_student(db, student_id)
    }
    out = []
    for skill in PROFILE_SKILLS:
        row = latest.get(skill)
        out.append(CatSkillOut(
            skill=skill,
            p_l=row.p_l if row else None,
            confidence=row.confidence if row else None,
        ))
    return out


def _top3(profile: list[CatSkillOut]) -> list[CatSkillOut]:
    """按 P(L) 升序取最弱 3 项；没有数据的技法排最后。"""
    ranked = sorted(
        profile,
        key=lambda x: (x.p_l is None, x.p_l if x.p_l is not None else 0.0),
    )
    return ranked[:3]


def report(db: Session, *, user_id: int, is_student: bool, record_id: int) -> dict:
    record = db.get(PracticeRecord, record_id)
    if record is None or record.source != "cat_test" or record.student_id is None:
        raise BusinessError(404, "摸底记录不存在")

    if is_student:
        own = _own_student_id(db, user_id)
        if own != record.student_id:
            raise BusinessError(403, "无权查看该记录")

    profile = _profile(db, record.student_id)
    return {
        "record_id": record.id,
        "student_id": record.student_id,
        "profile": profile,
        "top3": _top3(profile),
    }


def skip(db: Session, *, user_id: int) -> dict:
    """8.10：只给还没有 BKT 的技法写默认值，不覆盖已有画像。"""
    student_id = _own_student_id(db, user_id)
    created = 0
    for skill in PROFILE_SKILLS:
        if bkt_history_repo.get_latest_for_student_skill(db, student_id, skill) is not None:
            continue
        db.add(BktHistory(
            student_id=student_id,
            skill=skill,
            p_l=_SKIP_P_L,
            confidence=_SKIP_CONFIDENCE,
            source_type="cat_test",
            source_id=None,
        ))
        created += 1
    db.commit()

    profile = _profile(db, student_id)
    return {"created": created, "profile": profile, "top3": _top3(profile)}
```

```js
async function recordedWav(chunks, mimeType) {
  const raw = new Blob(chunks, { type: mimeType || "audio/webm" });
  const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  let decoded;
  try {
    decoded = await audioCtx.decodeAudioData(await raw.arrayBuffer());
  } finally {
    audioCtx.close().catch(() => {});
  }

  const frames = Math.max(1, Math.ceil(decoded.duration * WAV_SAMPLE_RATE));
  const OfflineCtx = window.OfflineAudioContext || window.webkitOfflineAudioContext;
  const offline = new OfflineCtx(1, frames, WAV_SAMPLE_RATE);
  const src = offline.createBufferSource();
  src.buffer = decoded;
  src.connect(offline.destination);
  src.start();
  const samples = (await offline.startRendering()).getChannelData(0);

  const buffer = new ArrayBuffer(44 + samples.length * 2);
  const view = new DataView(buffer);
  const writeStr = (offset, s) => {
    for (let i = 0; i < s.length; i++) view.setUint8(offset + i, s.charCodeAt(i));
  };
  writeStr(0, "RIFF");
  view.setUint32(4, 36 + samples.length * 2, true);
  writeStr(8, "WAVE");
  writeStr(12, "fmt ");
  view.setUint32(16, 16, true);
  view.setUint16(20, 1, true);
  view.setUint16(22, 1, true);
  view.setUint32(24, WAV_SAMPLE_RATE, true);
  view.setUint32(28, WAV_SAMPLE_RATE * 2, true);
  view.setUint16(32, 2, true);
  view.setUint16(34, 16, true);
  writeStr(36, "data");
  view.setUint32(40, samples.length * 2, true);
  for (let i = 0, o = 44; i < samples.length; i++, o += 2) {
    const s = Math.max(-1, Math.min(1, samples[i]));
    view.setInt16(o, s < 0 ? s * 0x8000 : s * 0x7fff, true);
  }
  return {
    blob: new Blob([buffer], { type: "audio/wav" }),
    duration: samples.length / WAV_SAMPLE_RATE
  };
}

async function pollAnalysis(taskId, onProgress) {
  const startedAt = Date.now();
  while (true) {
    if (Date.now() - startedAt > ANALYZE_TIMEOUT_MS) {
      throw new Error("分析超时，请稍后重试");
    }
    const s = await apiGet("/api/analyze/status/" + taskId);
    if (s.status === "failed") throw new Error(s.message || "分析失败");
    if (s.status === "done") return apiGet("/api/analyze/result/" + taskId);
    if (onProgress) onProgress(s);
    await new Promise(r => setTimeout(r, ANALYZE_POLL_MS));
  }
}
```
